    /**
     * 🎤 Process user comment and generate player responses
     */
    public void processUserComment(String userComment, String whitePlayer, String blackPlayer,
                                   String currentGamePhase, DialogueCallback callback) {

        Log.d(TAG, "🎤 Processing user comment: " + userComment);

        // Prevent spam comments
        long currentTime = System.currentTimeMillis();
        if (currentTime - lastUserCommentTime < USER_COMMENT_COOLDOWN) {
            Log.d(TAG, "⏰ Comment too soon, ignoring");
            return;
        }

        lastUserComment = userComment;
        lastUserCommentTime = currentTime;

        executorService.execute(() -> {
            try {
                // Analyze comment sentiment and topic
                CommentAnalysis analysis = analyzeUserComment(userComment);
                Log.d(TAG, "📊 Comment analysis: " + analysis.toString());

                // Choose which player should respond first (if any)
                String respondingPlayer = chooseRespondingPlayer(analysis, whitePlayer, blackPlayer);

                if (respondingPlayer != null) {
                    Log.d(TAG, "🎭 " + respondingPlayer + " will respond to user comment");

                    // Generate contextual response
                    String response = generateUserCommentResponse(
                            userComment, respondingPlayer, currentGamePhase, analysis);
                    if (response != null && !response.trim().isEmpty()) {
                        mainHandler.post(() -> {
                            // Trigger callback with special user interaction flag
                            callback.onDialogueGenerated(respondingPlayer, response);

                            // 🎵 FIXED: Call with correct number of parameters
                            String emotionalState = determineEmotionalResponseToComment(analysis, respondingPlayer);
                            speakDialogueWithPersonalityAndEmotion(respondingPlayer, response, emotionalState, null);

                            // Maybe trigger a follow-up from the other player
                            if (analysis.likelihood > 0.7f && !conversationInProgress) {
                                String otherPlayer = respondingPlayer.equals(whitePlayer) ? blackPlayer : whitePlayer;
                                scheduleUserCommentFollowUp(otherPlayer, userComment, response, callback);
                            }
                        });
                    }
                } else {
                    Log.d(TAG, "🤐 No player chose to respond to this comment");
                }

            } catch (Exception e) {
                Log.e(TAG, "❌ Error processing user comment", e);
                mainHandler.post(() -> callback.onError("Error processing user comment"));
            }
        });
    }

    /**
     * 🎭 Determine emotional response to user comment based on analysis and master personality
     */
    private String determineEmotionalResponseToComment(CommentAnalysis analysis, String respondingPlayer) {
        // Base emotional response on comment sentiment and master personality
        switch (analysis.sentiment) {
            case "positive":
                // Positive comments generally make masters pleased
                if (analysis.topic.equals("praise")) {
                    return "pleased";
                } else {
                    return "neutral";
                }

            case "negative":
                // Negative comments affect masters differently based on personality
                switch (respondingPlayer.toLowerCase()) {
                    case "fischer":
                        return "frustrated"; // Fischer doesn't tolerate criticism well
                    case "tal":
                        return "neutral";    // Tal stays positive

                        return "concerned";  // Kasparov takes criticism seriously
                    case "karpov":
                        return "neutral";    // Karpov stays diplomatic
                    default:
                        return "concerned";
                }

            default: // neutral sentiment
                if (analysis.topic.equals("question")) {
                    return "pleased";    // Masters generally like answering chess questions
                } else {
                    return "neutral";
                }
        }
    }

    /**
     * 🎯 Determine emotional state based on evaluation and context
     */
    private String determineEmotionalStateFromEvaluation(String player, Float currentEvaluation) {
        if (currentEvaluation == null) {
            Log.d(TAG, "📊 No evaluation available - using neutral emotion");
            return "neutral";
        }

        Float previousEval = lastPlayerEvaluations.get(player.toLowerCase());
        if (previousEval == null) {
            Log.d(TAG, "📊 No previous evaluation for " + player + " - using neutral emotion");
            return "neutral";
        }

        float evaluationChange = currentEvaluation - previousEval;
        Log.d(TAG, "📊 Evaluation change for " + player + ": " + previousEval + " → " + currentEvaluation + " (Δ" + evaluationChange + ")");

        // Determine if this change is good or bad for the current player
        // Note: evaluation is from white's perspective, so adjust for black
        boolean isWhitePlayer = isPlayerWhite(player);
        float adjustedChange = isWhitePlayer ? evaluationChange : -evaluationChange;

        Log.d(TAG, "📊 Adjusted change for " + player + " (" + (isWhitePlayer ? "white" : "black") + "): " + adjustedChange);

        String emotionalState = OpenAITTSService.determineEmotionalState(adjustedChange, currentEvaluation);
        Log.d(TAG, "🎭 Emotional state for " + player + ": " + emotionalState);

        return emotionalState;
    }

    /**
     * 🧹 Clean and personalize user response for specific master
     */
    private String cleanAndPersonalizeUserResponse(String response, String respondingPlayer) {
        if (response == null || response.trim().isEmpty()) {
            return getFallbackUserResponse(respondingPlayer, new CommentAnalysis());
        }

        String cleaned = response.trim();

        // Remove quotation marks if present
        if (cleaned.startsWith("\"") && cleaned.endsWith("\"")) {
            cleaned = cleaned.substring(1, cleaned.length() - 1);
        }

        // Remove meta-language that doesn't sound natural
        cleaned = cleaned.replaceAll("(?i)^(As .+, |Well, |Actually, |You know, )", "");
        cleaned = cleaned.replaceAll("(?i)^(I would say |I think |I believe )", "");

        // Add master-specific personality touches for user interactions
        return addUserInteractionPersonality(cleaned, respondingPlayer);
    }

    /**
     * ✨ Add personality touches specific to user interactions
     */
    private String addUserInteractionPersonality(String response, String master) {
        if (response == null || response.trim().isEmpty()) {
            return response;
        }

        // Add subtle personality touches that feel natural for user interactions
        switch (master.toLowerCase()) {
            case "tal":
                // Tal was known for being warm and engaging with fans
                if (response.toLowerCase().contains("move") && Math.random() > 0.8) {
                    return response + " Chess is such a beautiful game!";
                }
                break;

            case "fischer":
                // Fischer was more serious in interactions
                if (response.toLowerCase().contains("correct") && Math.random() > 0.9) {
                    return response + " Precision is everything.";
                }
                break;

            case "carlsen":
                // Carlsen is modern and more casual
                if (response.toLowerCase().contains("interesting") && Math.random() > 0.85) {
                    return response + " Always looking for practical chances!";
                }
                break;

            case "kasparov":
                // Kasparov was passionate about chess education
                if (response.toLowerCase().contains("position") && Math.random() > 0.85) {
                    return response + " This is what makes chess fascinating!";
                }
                break;
        }

        return response; // Return unchanged most of the time to avoid over-personalizing
    }

    /**
     * 🎛️ Enhanced comment analysis for better response triggering
     */
    private CommentAnalysis enhanceCommentAnalysis(String comment) {
        CommentAnalysis analysis = analyzeUserComment(comment);

        // Additional analysis for better response quality
        String lowerComment = comment.toLowerCase();

        // Detect chess-specific terms that increase response likelihood
        if (lowerComment.contains("sacrifice") || lowerComment.contains("gambit") ||
                lowerComment.contains("checkmate") || lowerComment.contains("endgame")) {
            analysis.likelihood += 0.2f;
            analysis.topic = "chess_technical";
        }

        // Detect questions more accurately
        if (lowerComment.startsWith("why") || lowerComment.startsWith("what") ||
                lowerComment.startsWith("how") || lowerComment.contains("?")) {
            analysis.topic = "question";
            analysis.likelihood += 0.3f;
        }

        // Detect emotional expressions
        if (lowerComment.contains("amazing") || lowerComment.contains("incredible") ||
                lowerComment.contains("brilliant")) {
            analysis.sentiment = "very_positive";
            analysis.likelihood += 0.25f;
        }

        // Cap likelihood at 1.0
        analysis.likelihood = Math.min(1.0f, analysis.likelihood);

        return analysis;
    }

    /**
     * 🧠 Analyze user comment for sentiment, topic, and response worthiness
     */
    private CommentAnalysis analyzeUserComment(String comment) {
        CommentAnalysis analysis = new CommentAnalysis();
        String lowerComment = comment.toLowerCase();

        // Determine topic
        if (lowerComment.contains("move") || lowerComment.contains("play")) {
            analysis.topic = "move_critique";
            analysis.likelihood = 0.8f;
        } else if (lowerComment.contains("good") || lowerComment.contains("nice") || lowerComment.contains("brilliant")) {
            analysis.topic = "praise";
            analysis.sentiment = "positive";
            analysis.likelihood = 0.9f;
        } else if (lowerComment.contains("bad") || lowerComment.contains("mistake") || lowerComment.contains("blunder")) {
            analysis.topic = "criticism";
            analysis.sentiment = "negative";
            analysis.likelihood = 0.9f;
        } else if (lowerComment.contains("question") || lowerComment.contains("why") || lowerComment.contains("what")) {
            analysis.topic = "question";
            analysis.likelihood = 0.85f;
        } else if (lowerComment.contains("hello") || lowerComment.contains("hi") || lowerComment.contains("watching")) {
            analysis.topic = "greeting";
            analysis.likelihood = 0.7f;
        } else if (lowerComment.contains("boring") || lowerComment.contains("slow") || lowerComment.contains("exciting")) {
            analysis.topic = "game_pace";
            analysis.likelihood = 0.6f;
        } else {
            analysis.topic = "general";
            analysis.likelihood = 0.4f; // Lower chance of response for unclear comments
        }

        // Determine sentiment if not already set
        if (analysis.sentiment.equals("neutral")) {
            if (lowerComment.contains("love") || lowerComment.contains("amazing") || lowerComment.contains("awesome")) {
                analysis.sentiment = "positive";
            } else if (lowerComment.contains("hate") || lowerComment.contains("terrible") || lowerComment.contains("awful")) {
                analysis.sentiment = "negative";
            }
        }

        return analysis;
    }

    /**
     * 🎯 Choose which player should respond to the user comment
     */
    private String chooseRespondingPlayer(CommentAnalysis analysis, String whitePlayer, String blackPlayer) {
        // Some masters are more likely to engage with spectators
        Map<String, Float> engagementLikelihood = new HashMap<>();
        engagementLikelihood.put("tal", 0.9f);        // Tal loved talking about chess
        engagementLikelihood.put("carlsen", 0.8f);    // Carlsen is modern and engaging
        engagementLikelihood.put("kasparov", 0.85f);  // Kasparov loves chess education
        engagementLikelihood.put("fischer", 0.3f);    // Fischer was less social
        engagementLikelihood.put("karpov", 0.6f);     // Karpov more diplomatic
        engagementLikelihood.put("kramnik", 0.5f);    // Kramnik thoughtful but reserved

        float whiteLikelihood = engagementLikelihood.getOrDefault(whitePlayer.toLowerCase(), 0.6f);
        float blackLikelihood = engagementLikelihood.getOrDefault(blackPlayer.toLowerCase(), 0.6f);

        // Adjust based on comment type
        if (analysis.topic.equals("praise") && analysis.sentiment.equals("positive")) {
            // More likely to respond to praise
            whiteLikelihood *= 1.3f;
            blackLikelihood *= 1.3f;
        } else if (analysis.topic.equals("criticism")) {
            // Fischer more likely to respond to criticism, others less so
            if ("fischer".equals(whitePlayer.toLowerCase())) whiteLikelihood *= 1.5f;
            if ("fischer".equals(blackPlayer.toLowerCase())) blackLikelihood *= 1.5f;
            else {
                whiteLikelihood *= 0.7f;
                blackLikelihood *= 0.7f;
            }
        }

        // Random selection weighted by likelihood
        if (Math.random() < whiteLikelihood && Math.random() > blackLikelihood) {
            return whitePlayer;
        } else if (Math.random() < blackLikelihood) {
            return blackPlayer;
        }

        return null; // No response
    }

                                               String gamePhase, CommentAnalysis analysis) {

        String playerName = modelManager.getMasterDisplayName(respondingPlayer);

        String prompt = createUserCommentResponsePrompt(
                respondingPlayer, userComment, gamePhase, analysis);

        try {
            String response = openAIService.getChatCompletion(prompt,
                    "A spectator just said: \"" + userComment + "\". Respond naturally as " + playerName + ".");

            return cleanAndPersonalizeUserResponse(response, respondingPlayer);

        } catch (Exception e) {
            Log.e(TAG, "Error generating user comment response", e);
            return getFallbackUserResponse(respondingPlayer, analysis);
        }
    }

    /**
     * 🎭 Create response prompt for user comments
     */
    private String createUserCommentResponsePrompt(String master, String userComment,
                                                   String gamePhase, CommentAnalysis analysis) {
        String masterName = modelManager.getMasterDisplayName(master);
        StringBuilder prompt = new StringBuilder();

        prompt.append("You are ").append(masterName).append(" playing a chess game. ");
        prompt.append("A spectator watching your game just said: \"").append(userComment).append("\"\n\n");

        // Add personality context
        String personalityNote = getPersonalityNote(master);
        prompt.append(personalityNote).append("\n\n");

        // Add context-specific guidance
        switch (analysis.topic) {
            case "praise":
                prompt.append("The spectator is praising something about the game. ");
                prompt.append("Respond graciously but stay focused on the chess. ");
                break;
            case "criticism":
                prompt.append("The spectator is criticizing something. ");
                if ("fischer".equals(master.toLowerCase())) {
                    prompt.append("As Fischer, you might respond with intensity about chess truth. ");
                } else {
                    prompt.append("Respond diplomatically while defending good chess. ");
                }
                break;
            case "question":
                prompt.append("The spectator asked a question. ");
                prompt.append("Answer briefly while staying focused on your game. ");
                break;
            case "greeting":
                prompt.append("The spectator is greeting you or acknowledging they're watching. ");
                prompt.append("Acknowledge them briefly and warmly. ");
                break;
            default:
                prompt.append("Respond naturally to their comment. ");
                break;
        }

        prompt.append("\n\nKeep your response to 1-2 sentences. ");
        prompt.append("Sound like you're actually playing chess and can't chat too long. ");
        prompt.append("Be authentic to ").append(masterName).append("'s personality.");

        return prompt.toString();
    }

    /**
     * 🎭 Create conversational response
     */
    private void generateConversationResponse(String responder, String originalSpeaker,
                                              String triggerStatement, String context, DialogueCallback callback) {

        executorService.execute(() -> {
            try {
                conversationInProgress = true;
                conversationTurnCount++;

                String responderName = modelManager.getMasterDisplayName(responder);
                String originalSpeakerName = modelManager.getMasterDisplayName(originalSpeaker);

                String conversationPrompt = createConversationalPrompt(responder, String.format(
                        "%s just said: \"%s\"\n\n" +
                                "As %s, respond naturally to what they said. This is a conversation between two chess masters. " +
                                "Keep your response brief (1-2 sentences) and authentic to your personality. " +
                                "You might agree, disagree, add your own perspective, or ask a question. " +
                                "Sound like you're actually talking to %s, not giving a lecture. " +
                                "Examples: 'I agree!' or 'That's one way to look at it' or 'You always were aggressive' " +
                                "BE CONVERSATIONAL, not analytical!",
                        originalSpeakerName, triggerStatement, responderName, originalSpeakerName
                ));

                String response = cleanAndPersonalize(
                        openAIService.getChatCompletion(conversationPrompt, "Generate a conversational response"),
                        responder
                );

                // Record this turn
                conversationHistory.add(new ConversationTurn(responder, response, context));
                lastSpeaker = responder;
                lastStatement = response;

                mainHandler.post(() -> {
                    callback.onDialogueGenerated(responder, response);

                    // 🎵 SPEAK the response with personality voice!
                    speakDialogueWithPersonality(responder, response);

                    // Check if this conversation should continue
                    if (conversationTurnCount < MAX_CONVERSATION_TURNS && shouldContinueConversation(response)) {
                        scheduleConversationResponse(originalSpeaker, responder, response, "conversation_continue", callback);
                    } else {
                        // End the conversation
                        conversationInProgress = false;
                        conversationTurnCount = 0;
                        callback.onConversationComplete(responder, response);
                        Log.d(TAG, "🎭 Conversation completed naturally");
                    }
                });

            } catch (Exception e) {
                Log.e(TAG, "Error generating conversation response", e);
                conversationInProgress = false;
                conversationTurnCount = 0;
                mainHandler.post(() -> callback.onError("Failed to generate conversation response"));
            }
        });
    }

    /**
     * ⏰ Schedule a conversation response with natural timing
     */
    private void scheduleConversationResponse(String responder, String originalSpeaker,
                                              String triggerStatement, String context, DialogueCallback callback) {

        Log.d(TAG, "⏰ Scheduling conversation response from " + responder);

        mainHandler.postDelayed(() -> {
            if (!conversationInProgress && conversationTurnCount >= MAX_CONVERSATION_TURNS) {
                Log.d(TAG, "⏸️ Conversation limit reached, not scheduling response");
                return;
            }

            callback.onConversationStarted(responder, triggerStatement);
            generateConversationResponse(responder, originalSpeaker, triggerStatement, context, callback);

        }, CONVERSATION_DELAY_MS);
    }

    /**
     * 🎯 Determine if a statement should trigger a response
     */
    private boolean shouldTriggerResponse(String statement) {
        if (statement == null || statement.trim().isEmpty()) {
            return false;
        }

        String lowerStatement = statement.toLowerCase();

        // Conversation triggers
        String[] triggers = {
                "interesting", "bold", "what do you think", "agree", "disagree",
                "remember", "always", "never", "brilliant", "mistake",
                "classic", "typical", "style", "approach", "strategy",
                "battle", "fight", "challenge", "ready", "see what", "make of this"
        };

        for (String trigger : triggers) {
            if (lowerStatement.contains(trigger)) {
                return true;
            }
        }

        // Questions always trigger responses
        if (lowerStatement.contains("?")) {
            return true;
        }

        // Random chance for natural conversation flow
        return Math.random() > 0.7; // 30% chance of spontaneous response
    }

    /**
     * 🎲 Choose who comments on a move
     */
    private String chooseCommentator(String playerWhoMoved, String whitePlayer, String blackPlayer, int moveNumber) {
        // Early game: more likely to comment on own moves
        // Later game: more likely to comment on opponent moves

        if (moveNumber <= 10) {
            // Early game - 70% chance player comments on own move
            return Math.random() > 0.3 ? playerWhoMoved :
                    (playerWhoMoved.equals(whitePlayer) ? blackPlayer : whitePlayer);
        } else {
            // Later game - 60% chance opponent comments
            return Math.random() > 0.4 ?
                    (playerWhoMoved.equals(whitePlayer) ? blackPlayer : whitePlayer) : playerWhoMoved;
        }
    }

    /**
     * 🔄 Check if conversation should continue
     */
    private boolean shouldContinueConversation(String lastResponse) {
        if (lastResponse == null) return false;

        String lower = lastResponse.toLowerCase();

        // Don't continue if response is very short or seems final
        if (lastResponse.length() < 20) return false;

        // Continue if it seems like there's more to discuss
        return lower.contains("but") || lower.contains("however") ||
                lower.contains("though") || lower.contains("?") ||
                lower.contains("what about") || lower.contains("consider");
    }

    /**
     * 🧹 Clean and personalize responses
     */
    private String cleanAndPersonalize(String response, String master) {
        if (response == null || response.trim().isEmpty()) {
            return getFallbackResponse(master);
        }

        String cleaned = response.trim();

        // Remove quotation marks if present
        if (cleaned.startsWith("\"") && cleaned.endsWith("\"")) {
            cleaned = cleaned.substring(1, cleaned.length() - 1);
        }

        // Remove meta-language
        cleaned = cleaned.replaceAll("(?i)^(Let me |I would |I think |I believe |As .+, )", "");
        cleaned = cleaned.replaceAll("(?i)^(This |Looking at |Analyzing )", "");

        // Add personality touches
        return addPersonalityTouch(cleaned, master);
    }


    /**
     * ✨ FIXED: Add personality-specific touches WITHOUT excessive repetition
     */
    private String addPersonalityTouch(String response, String master) {
        if (response == null || response.trim().isEmpty()) {
            return response;
        }

        // MUCH MORE SELECTIVE: Only add personality touches rarely and contextually
        switch (master.toLowerCase()) {
            case "tal":
                // Only add "Beautiful chess!" for truly brilliant tactical content
                if (response.toLowerCase().contains("sacrifice") &&
                        response.toLowerCase().contains("brilliant") &&
                        !response.toLowerCase().contains("beautiful") &&
                        Math.random() > 0.95) { // Only 5% chance
                    return response + " Beautiful chess!";
                }
                break;

            case "fischer":
                // Only add perfectionist touch for accuracy-related content
                if (response.toLowerCase().contains("precise") &&
                        response.toLowerCase().contains("best") &&
                        Math.random() > 0.95) { // Only 5% chance
                    return response + " Perfection is essential.";
                }
                break;

            case "kasparov":
                // Only add fighting spirit for battle-related content
                if (response.toLowerCase().contains("fight") &&
                        response.toLowerCase().contains("battle") &&
                        Math.random() > 0.95) { // Only 5% chance
                    return response + " We must fight for every advantage!";
                }
                break;

            // Leave other masters clean to prevent any repetition
        }

        return response; // Return unchanged response most of the time
    }

                {"kasparov", "The battle continues!"},
                {"karpov", "Patience will decide this game."},
                {"kramnik", "Solid play is key here."}
        };

        for (String[] fallback : fallbacks) {
            if (fallback[0].equals(master.toLowerCase())) {
                return fallback[1];
            }
        }

        return "This should be interesting...";
    }

    /**
     * 🏆 Extract winner from result string
     */
    private String extractWinner(String result) {
        if (result == null) return null;

        String lower = result.toLowerCase();
        if (lower.contains("white wins") || lower.contains("black wins")) {
            return lower.contains("white") ? "white" : "black";
        }

        // Try to extract from result patterns
        if (lower.contains("tal") && lower.contains("wins")) return "tal";
        if (lower.contains("fischer") && lower.contains("wins")) return "fischer";
        if (lower.contains("kasparov") && lower.contains("wins")) return "kasparov";

        return null; // Draw or unclear
    }

    /**
     * 🔄 Reset conversation state
     */
    public void resetConversation() {
        conversationHistory.clear();
        conversationInProgress = false;
        conversationTurnCount = 0;
        lastSpeaker = "";
        lastStatement = "";

        // Reset TTS state
        isSpeaking = false;
        currentlySpeaking = "";

        Log.d(TAG, "🔄 Conversation state and TTS state reset");
    }

    /**
     * 🛑 NEW: Stop current speech (useful for interruptions or game pause)
     */
    public void stopCurrentSpeech() {
        Log.d(TAG, "🛑 Stopping current speech");
        if (ttsService != null && isSpeaking) {
            ttsService.interrupt();
            isSpeaking = false;
            currentlySpeaking = "";
        }
    }

    /**
     * ℹ️ NEW: Get current speaking state
     */
    public boolean isSpeaking() {
        return isSpeaking;
    }

    public String getCurrentlySpeaking() {
        return currentlySpeaking;
    }

    /**
     * 🧹 Cleanup resources
     */
    public void cleanup() {
        stopCurrentSpeech(); // Stop any ongoing speech

        if (executorService != null && !executorService.isShutdown()) {
            executorService.shutdown();
        }
        resetConversation();
        Log.d(TAG, "🧹 AIDialogueManager cleaned up");
    }
